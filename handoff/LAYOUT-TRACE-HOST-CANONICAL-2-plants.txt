# LAYOUT-TRACE-HOST-CANONICAL-2 plants (cargo xtask plants; docs/plants.md). Each command traces every regen
# probe (regen.sh --trace-only) against a RAM copy of the role's fast products (built from this tree); the tracer
# under test is the plant copy's. The brief's two plants: an IFUNC candidate following the host's CPU (lthc2-cpu-host)
# and the host's locale data read (lthc2-locale-host); then the other host inputs and the syscall-stop fix.
# lthc2-cpu-host: the CPU entry runs its own code (the host's CPUID) instead of the virtual CPU's answer.
lthc2-cpu-host | tools/hottext/trace_gdb.py | lit:answer(emul[pc], cls) ==> pass | d=$(mktemp -d /mnt/zram/claude/lthc2-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /mnt/zram/claude/LAYOUT-TRACE-HOST-CANONICAL-2/prod --trace-only $d | msg:ran the same functions: the program did not see the virtual CPU
# lthc2-locale-host: the locale archive is the host's, not the tree's en_US.UTF-8 archive.
lthc2-locale-host | tools/hottext/canon_host.sh | lit:/usr/lib/locale/locale-archive=$a: ==> | d=$(mktemp -d /mnt/zram/claude/lthc2-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /mnt/zram/claude/LAYOUT-TRACE-HOST-CANONICAL-2/prod --trace-only $d | msg:read host files the tree does not supply.*/usr/lib/locale/locale-archive
# lthc2-nscache-host: the name-service cache region is the host's /dev/shm file (create or attach by host state).
lthc2-nscache-host | tools/hottext/canon_host.sh | lit::/dev/shm/liella-nscache-$u=@run/nscache ==> | d=$(mktemp -d /mnt/zram/claude/lthc2-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /mnt/zram/claude/LAYOUT-TRACE-HOST-CANONICAL-2/prod --trace-only $d | msg:read host files the tree does not supply.*/dev/shm/liella-nscache-[0-9]+
# lthc2-resolv-host: the resolver configuration is the host's.
lthc2-resolv-host | tools/hottext/canon_host.sh | lit:for f in nsswitch.conf resolv.conf host.conf gai.conf; ==> for f in nsswitch.conf host.conf gai.conf; | d=$(mktemp -d /mnt/zram/claude/lthc2-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /mnt/zram/claude/LAYOUT-TRACE-HOST-CANONICAL-2/prod --trace-only $d | msg:read host files the tree does not supply.*/etc/resolv.conf
# lthc2-syscall-bp: the breakpoint on a system call's return address is kept at its entry stop (the program ends there).
lthc2-syscall-bp | tools/hottext/trace_gdb.py | perl:s/(count the function at the return\.\n\s*)bp\.delete\(\)/$1pass/ | d=$(mktemp -d /mnt/zram/claude/lthc2-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /mnt/zram/claude/LAYOUT-TRACE-HOST-CANONICAL-2/prod --trace-only $d | msg:p_cold: library functions named in the source never ran
