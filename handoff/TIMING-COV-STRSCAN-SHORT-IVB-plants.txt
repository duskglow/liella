# TIMING-COV-STRSCAN-SHORT-IVB plants: the strchr / strchrnul SSE2 head blocks 1-3 (fall-through misses,
# out-of-line hit tails) in csys/src/machine/scan.S. machine_scanners runs each kernel on both scan levels
# (SSE2, and AVX2 on the build host); spec §4.5 (strchr/strchrnul).
# strchr: block 1's tail returns one block early (offset 0 instead of 16).
strchr-t16-offset | csys/src/machine/scan.S | perl:s/(liella_machine_strchr:.*?\.Lschr_t16:.*?)leaq 16\(%rdi,%rcx\), %rax/$1leaq (%rdi,%rcx), %rax/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchr: block 2's hit goes to block 3's tail (result 16 bytes late).
strchr-blk2-tail | csys/src/machine/scan.S | perl:s/(liella_machine_strchr:.*?)jnz \.Lschr_t32/$1jnz .Lschr_t48/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchr: block 2's tail off by one.
strchr-t32-offset | csys/src/machine/scan.S | perl:s/(liella_machine_strchr:.*?\.Lschr_t32:.*?)leaq 32\(%rdi,%rcx\), %rax/$1leaq 31(%rdi,%rcx), %rax/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchr: block 3's tail swaps c and NUL (NULL for a found c, a pointer for the terminator).
strchr-t48-null | csys/src/machine/scan.S | perl:s/(liella_machine_strchr:.*?\.Lschr_t48:.*?)jnc \.Lschr_snull/$1jc .Lschr_snull/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchrnul: block 1's tail one block early.
strchrnul-t16-offset | csys/src/machine/scan.S | perl:s/(liella_machine_strchrnul:.*?\.Lschrnul_t16:.*?)leaq 16\(%rdi,%rdx\), %rax/$1leaq (%rdi,%rdx), %rax/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchrnul: block 1's hit goes to block 2's tail.
strchrnul-blk1-tail | csys/src/machine/scan.S | perl:s/(liella_machine_strchrnul:.*?)jnz \.Lschrnul_t16/$1jnz .Lschrnul_t32/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchrnul: block 2's tail off by one.
strchrnul-t32-offset | csys/src/machine/scan.S | perl:s/(liella_machine_strchrnul:.*?\.Lschrnul_t32:.*?)leaq 32\(%rdi,%rdx\), %rax/$1leaq 33(%rdi,%rdx), %rax/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
# strchrnul: block 3's tail returns block 2's position.
strchrnul-t48-offset | csys/src/machine/scan.S | perl:s/(liella_machine_strchrnul:.*?\.Lschrnul_t48:.*?)leaq 48\(%rdi,%rdx\), %rax/$1leaq 32(%rdi,%rdx), %rax/s | cargo test -p liella-boundary --test machine_scanners | msg:string scanner postcondition failed|strchr.*(FAILED|panicked)
