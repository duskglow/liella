# LAYOUT-TRACE-HOST-CANONICAL-3 plants (cargo xtask plants; docs/plants.md), run on a Zen 1 host (RAM under
# /dev/shm/claude). The five plants of -2 (handoff/LAYOUT-TRACE-HOST-CANONICAL-2-plants.txt) with this host's RAM
# paths: each command traces every regen probe (regen.sh --trace-only) against a RAM copy of the role's fast
# products (built from this tree); the tracer under test is the plant copy's. Then the runner's environment
# hand-off, caught by the xtask unit test this run fixed.
# lthc3-cpu-host: the CPU entry runs its own code (the host's CPUID) instead of the virtual CPU's answer.
lthc3-cpu-host | tools/hottext/trace_gdb.py | lit:answer(emul[pc], cls) ==> pass | d=$(mktemp -d /dev/shm/claude/lthc3-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /dev/shm/claude/LAYOUT-TRACE-HOST-CANONICAL-3/prod --trace-only $d | msg:ran the same functions: the program did not see the virtual CPU
# lthc3-locale-host: the locale archive is the host's, not the tree's en_US.UTF-8 archive.
lthc3-locale-host | tools/hottext/canon_host.sh | lit:/usr/lib/locale/locale-archive=$a: ==> | d=$(mktemp -d /dev/shm/claude/lthc3-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /dev/shm/claude/LAYOUT-TRACE-HOST-CANONICAL-3/prod --trace-only $d | msg:read host files the tree does not supply.*/usr/lib/locale/locale-archive
# lthc3-nscache-host: the name-service cache region is the host's /dev/shm file (create or attach by host state).
lthc3-nscache-host | tools/hottext/canon_host.sh | lit::/dev/shm/liella-nscache-$u=@run/nscache ==> | d=$(mktemp -d /dev/shm/claude/lthc3-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /dev/shm/claude/LAYOUT-TRACE-HOST-CANONICAL-3/prod --trace-only $d | msg:read host files the tree does not supply.*/dev/shm/liella-nscache-[0-9]+
# lthc3-resolv-host: the resolver configuration is the host's.
lthc3-resolv-host | tools/hottext/canon_host.sh | lit:for f in nsswitch.conf resolv.conf host.conf gai.conf; ==> for f in nsswitch.conf host.conf gai.conf; | d=$(mktemp -d /dev/shm/claude/lthc3-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /dev/shm/claude/LAYOUT-TRACE-HOST-CANONICAL-3/prod --trace-only $d | msg:read host files the tree does not supply.*/etc/resolv.conf
# lthc3-syscall-bp: the breakpoint on a system call's return address is kept at its entry stop (the program ends there).
lthc3-syscall-bp | tools/hottext/trace_gdb.py | perl:s/(count the function at the return\.\n\s*)bp\.delete\(\)/$1pass/ | d=$(mktemp -d /dev/shm/claude/lthc3-pt.XXXXXX) && rmdir $d && bash tools/hottext/regen.sh . --products /dev/shm/claude/LAYOUT-TRACE-HOST-CANONICAL-3/prod --trace-only $d | msg:p_cold: library functions named in the source never ran
# lthc3-runner-env: the runner passes the program's assignments to the tracer's own environment, unprefixed
# (the program would then get no LANG/TZ from the trace and the caller's environment would reach gdb).
lthc3-runner-env | tools/hottext/runner.sh | lit:*=*) envs+=("HOT_TEXT_ENV_$1");; ==> *=*) envs+=("$1");; | cargo test -p xtask --bin xtask -- hottext::tests::regen_runner_runs_traces_together_and_keeps_their_order | msg:regen_runner_runs_traces_together_and_keeps_their_order.*(FAILED|panicked)|names\.1 is not trace p1
# lthc3-absent-keep: prune.py --absent keeps the allow-list line of a key that left the list (a replaced list then fails the release build).
lthc3-absent-keep | tools/hottext/prune.py | lit:or key in listed: ==> or True: | cargo test -p xtask --bin xtask -- hottext::tests::regen_replace_drops_the_allow_list_lines_of_keys_it_drops | msg:kept a stale line or lost a live one
# lthc3-absent-guard: regen.sh prunes the allow-list on every merge instead of --replace only (the train's merge would rewrite the hand-written file).
lthc3-absent-guard | tools/hottext/regen.sh | lit:if [ $replace = 1 ] && [ -f "$absent" ]; then ==> if [ -f "$absent" ]; then | cargo test -p xtask --bin xtask -- hottext::tests::regen_replace_drops_the_allow_list_lines_of_keys_it_drops | msg:no --replace allow-list prune
# lthc3-alias-unsorted: the names at one address keep nm's order (the caller's locale decides it).
lthc3-alias-unsorted | tools/hottext/trace_gdb.py | lit:names = {a: sorted(ns) for a, ns in names.items()} ==> names = dict(names) | cargo test -p xtask --bin xtask -- hottext::tests::trace_writes_the_names_at_one_address_in_byte_order | msg:does not sort the names at one address
# lthc3-nm-locale: nm runs in the caller's locale.
lthc3-nm-locale | tools/hottext/trace_gdb.py | lit:check=True, env=nm_env).stdout ==> check=True).stdout | cargo test -p xtask --bin xtask -- hottext::tests::trace_writes_the_names_at_one_address_in_byte_order | msg:runs nm in the caller.s locale
